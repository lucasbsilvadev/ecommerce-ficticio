from pathlib import Path

IGNORE_DIRS = {
    ".venv", "venv", ".env", "env",
    "__pycache__", ".pytest_cache", ".mypy_cache", ".ruff_cache",
    ".git", ".idea", ".vscode",
    "node_modules", "htmlcov",
}
IGNORE_FILES = {
    ".coverage", "coverage.xml", "snapshot.txt", "pytest-output.txt",
}
IGNORE_EXTS = {".pyc", ".pyo", ".pyd", ".log"}

def ignored(p: Path) -> bool:
    if p.name in IGNORE_DIRS or p.name in IGNORE_FILES:
        return True
    if p.suffix in IGNORE_EXTS:
        return True
    return False

def tree(root: Path, prefix: str = "") -> list[str]:
    lines = []
    items = sorted(root.iterdir(), key=lambda p: (not p.is_dir(), p.name.lower()))
    for item in items:
        if ignored(item):
            continue
        if item.is_dir():
            lines.append(f"{prefix}+---{item.name}")
            lines.extend(tree(item, prefix + "|   "))
        else:
            lines.append(f"{prefix}|   {item.name}")
    return lines

def dump(root: Path) -> str:
    parts = ["=== ESTRUTURA ===", "Listagem de caminhos de pasta", ""]
    parts.extend(tree(root))
    parts.append("")
    parts.append("")
    parts.append("=== CONTEUDO DOS ARQUIVOS ===")
    parts.append("")

    for path in sorted(root.rglob("*")):
        if path.is_dir() or ignored(path):
            continue
        if any(part in IGNORE_DIRS for part in path.parts):
            continue
        rel = path.relative_to(root)
        parts.append("=" * 60)
        parts.append(f"FILE: {rel}")
        parts.append("=" * 60)
        try:
            parts.append(path.read_text(encoding="utf-8"))
        except UnicodeDecodeError:
            parts.append("[binário — ignorado]")
        parts.append("")

    return "\n".join(parts)

if __name__ == "__main__":
    out = dump(Path("."))
    Path("snapshot.txt").write_text(out, encoding="utf-8")
    print(f"snapshot.txt gerado ({out.count(chr(10))} linhas)")

import { useState } from "react";
import { CartProvider, useCart } from "./features/cart/CartContext";
import AppLayout from "./components/layout/AppLayout";
import CatalogPage from "./features/catalog/pages/CatalogPage";
import ProductPage from "./features/catalog/pages/ProductPage";
import CartPage from "./features/cart/pages/CartPage";
import CheckoutPage from "./features/order/pages/CheckoutPage";
import ReviewPage from "./features/order/pages/ReviewPage";
import Preloader from "./components/ui/Preloader";
import TermosPage from "./features/legal/pages/TermosPage";
import PrivacidadePage from "./features/legal/pages/PrivacidadePage";
import ComoFuncionaPage from "./features/legal/pages/ComoFuncionaPage";

const LAYOUT_ROUTES = new Set(["catalog", "termos", "privacidade", "como-funciona"]);

function AppRoutes() {
  const [route, setRoute] = useState({ name: "catalog" });
  const { add, count } = useCart();

  function goPage(name) { setRoute({ name }); }
  function goCatalog() { setRoute({ name: "catalog" }); }
  function goProduct(produto) { setRoute({ name: "product", produto }); }
  function goCart() { setRoute({ name: "cart" }); }
  function goCheckout() { setRoute({ name: "checkout", data: route.data || {} }); }
  function goReview(data) { setRoute({ name: "review", data }); }

  function renderRoute() {
    switch (route.name) {
      case "catalog":
        return <CatalogPage onOpenProduct={goProduct} />;
      case "termos":
        return <TermosPage />;
      case "privacidade":
        return <PrivacidadePage />;
      case "como-funciona":
        return <ComoFuncionaPage />;
      case "product":
        return (
          <ProductPage
            produto={route.produto}
            onBack={goCatalog}
            onAdd={(p, qty) => { add(p, qty); goCart(); }}
          />
        );
      case "cart":
        return <CartPage onBack={goCatalog} onCheckout={goCheckout} />;
      case "checkout":
        return (
          <CheckoutPage
            initial={route.data}
            onBack={goCart}
            onNext={goReview}
          />
        );
      case "review":
        return (
          <ReviewPage
            data={route.data}
            onBack={() => setRoute({ name: "checkout", data: route.data })}
            onClear={goCatalog}
          />
        );
      default:
        return null;
    }
  }

  if (LAYOUT_ROUTES.has(route.name)) {
    return (
      <AppLayout
        current={route.name}
        onNavigate={goPage}
        onOpenCart={goCart}
        cartCount={count}
      >
        {renderRoute()}
      </AppLayout>
    );
  }

  return renderRoute();
}

export default function App() {
  return (
    <CartProvider>
      <Preloader />
      <AppRoutes />
    </CartProvider>
  );
}

import { useState } from "react";
import { ShoppingCart, Menu, X } from "lucide-react";
import "./Navbar.css";

const NAV_LINKS = [
  { key: "catalog", label: "Catálogo" },
  { key: "como-funciona", label: "Como funciona" },
  { key: "termos", label: "Termos" },
  { key: "privacidade", label: "Privacidade" },
];

export default function Navbar({ current, onNavigate, onOpenCart, cartCount }) {
  const [menuOpen, setMenuOpen] = useState(false);

  function handleNav(key) {
    setMenuOpen(false);
    onNavigate?.(key);
  }

  return (
    <header className="navbar">
      <button
        type="button"
        className="navbar__icon-btn navbar__menu-btn"
        onClick={() => setMenuOpen((v) => !v)}
        aria-label={menuOpen ? "Fechar menu" : "Abrir menu"}
        aria-expanded={menuOpen}
      >
        {menuOpen ? <X size={22} /> : <Menu size={22} />}
      </button>

      <button
        type="button"
        className="navbar__brand"
        onClick={() => handleNav("catalog")}
        aria-label="Ir para o catálogo"
      >
        <span className="navbar__brand-mark">GELADA</span>
        <span className="navbar__brand-sub">Depósito de Bebidas</span>
      </button>

      <nav className="navbar__nav" aria-label="Navegação principal">
        {NAV_LINKS.map((link) => (
          <button
            key={link.key}
            type="button"
            className={`navbar__nav-link ${current === link.key ? "is-current" : ""}`}
            onClick={() => handleNav(link.key)}
            aria-current={current === link.key ? "page" : undefined}
          >
            {link.label}
          </button>
        ))}
      </nav>

      <button
        type="button"
        className="navbar__icon-btn navbar__cart-btn"
        onClick={onOpenCart}
        aria-label={`Abrir carrinho (${cartCount} itens)`}

        <ShoppingCart size={22} />

          <span className="navbar__cart-badge">{cartCount}</span>

import { createContext, useContext, useMemo, useState, useEffect } from "react";

const CartContext = createContext(null);
const STORAGE_KEY = "deposito-cart-v1";

function loadCart() {
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    if (!raw) return [];
    const parsed = JSON.parse(raw);
    return Array.isArray(parsed) ? parsed : [];
  } catch {
    return [];
  }
}

export function CartProvider({ children }) {
  const [items, setItems] = useState(loadCart);

  useEffect(() => {
    try { localStorage.setItem(STORAGE_KEY, JSON.stringify(items)); } catch { /* ignore */ }
  }, [items]);

  const api = useMemo(() => ({
    items,
    count: items.reduce((s, i) => s + i.qty, 0),
    subtotal: items.reduce((s, i) => s + i.qty * i.precoNum, 0),
    add(produto, qty = 1) {
      setItems((cur) => {
        const idx = cur.findIndex((i) => i.nome === produto.nome);
        if (idx >= 0) {
          const copy = [...cur];
          copy[idx] = { ...copy[idx], qty: copy[idx].qty + qty };
          return copy;
        }
        return [...cur, {
          nome: produto.nome,
          preco: produto.preco,
          precoNum: parseFloat(produto.preco.replace(",", ".")),
          imagem: produto.imagem,
          qty,
        }];
      });
    },
    setQty(nome, qty) {
      setItems((cur) =>
        cur.map((i) => i.nome === nome ? { ...i, qty: Math.max(1, qty) } : i)
      );
    },
    remove(nome) {
      setItems((cur) => cur.filter((i) => i.nome !== nome));
    },
    clear() { setItems([]); },
  }), [items]);

  return <CartContext.Provider value={api}>{children}</CartContext.Provider>;
}

export function useCart() {
  const ctx = useContext(CartContext);
  if (!ctx) throw new Error("useCart must be used within CartProvider");
  return ctx;
}

import { useState } from "react";
import Button from "../../../components/ui/Button";
import "./ProductPage.css";

export default function ProductPage({ produto, onBack, onAdd }) {
  const [qty, setQty] = useState(1);
  const [added, setAdded] = useState(false);
  const [imgError, setImgError] = useState(false);

  if (!produto) return null;

  function handleAdd() {
    onAdd(produto, qty);
    setAdded(true);
    setTimeout(() => setAdded(false), 1500);
  }

  return (
    <div className="product-page">
      <button className="product-page__back" onClick={onBack} aria-label="Voltar ao catálogo">
        ← Voltar
      </button>

      <div className="product-page__image-wrap">
        {imgError || !produto.imagem ? (
          <div className="product-page__image-fallback" aria-hidden="true">🍺</div>
        ) : (
          <img
            src={produto.imagem}
            alt={produto.nome}
            className="product-page__image"
            onError={() => setImgError(true)}
          />
        )}
      </div>

      <div className="product-page__info">
        {produto.categoria && <span className="product-page__badge">{produto.categoria}</span>}
        <h1 className="product-page__name">{produto.nome}</h1>
        <p className="product-page__price">R$ {produto.preco}</p>

        <div className="product-page__qty">
          <span className="product-page__qty-label">Quantidade</span>
          <div className="product-page__qty-controls">
            <button
              type="button"
              onClick={() => setQty((q) => Math.max(1, q - 1))}
              aria-label="Diminuir quantidade"
            >−</button>
            <span aria-live="polite">{qty}</span>
            <button
              type="button"
              onClick={() => setQty((q) => q + 1)}
              aria-label="Aumentar quantidade"
            >+</button>
          </div>
        </div>

        <Button variant="primary" size="lg" fullWidth onClick={handleAdd}>
          {added ? "✓ Adicionado" : "Adicionar ao pedido"}
        </Button>
      </div>
    </div>
  );
}

import { useState } from "react";
import Button from "../../../components/ui/Button";
import "./CheckoutPage.css";

const PAYMENT_OPTIONS = [
  { value: "dinheiro", label: "Dinheiro" },
  { value: "pix", label: "Pix" },
  { value: "cartao", label: "Cartão (maquininha na entrega)" },
];

export default function CheckoutPage({ onBack, onNext, initial }) {
  const [step, setStep] = useState(0);
  const [data, setData] = useState(initial || {
    endereco: "",
    referencia: "",
    pagamento: "",
  });
  const [error, setError] = useState("");

  function update(field, value) {
    setData((d) => ({ ...d, [field]: value }));
    setError("");
  }

  function next() {
    if (step === 0 && !data.endereco.trim()) {
      setError("Informe o endereço para entrega.");
      return;
    }
    if (step === 2 && !data.pagamento) {
      setError("Escolha a forma de pagamento.");
      return;
    }
    if (step < 2) setStep(step + 1);
    else onNext(data);
  }

  function back() {
    if (step === 0) onBack();
    else setStep(step - 1);
    setError("");
  }

  return (
    <div className="checkout">
      <header className="checkout__header">
        <button className="checkout__back" onClick={back} aria-label="Voltar">←</button>
        <div>
          <p className="checkout__step">Etapa {step + 1} de 3</p>
          <h1>
            {step === 0 && "Endereço de entrega"}
            {step === 1 && "Ponto de referência"}
            {step === 2 && "Forma de pagamento"}
          </h1>
        </div>
      </header>

      <div className="checkout__body">
        {step === 0 && (
          <label className="checkout__field">
            <span>Rua, número, bairro</span>
            <textarea
              rows={3}
              value={data.endereco}
              onChange={(e) => update("endereco", e.target.value)}
              placeholder="Ex.: Rua das Flores, 123 — Centro"
              autoFocus
            />
          </label>
        )}

        {step === 1 && (
          <label className="checkout__field">
            <span>Ponto de referência (opcional)</span>
            <textarea
              rows={3}
              value={data.referencia}
              onChange={(e) => update("referencia", e.target.value)}
              placeholder="Ex.: próximo ao mercado, portão azul"
              autoFocus
            />
          </label>
        )}

        {step === 2 && (
          <fieldset className="checkout__payments">
            <legend className="visually-hidden">Forma de pagamento</legend>
            {PAYMENT_OPTIONS.map((opt) => (
              <label key={opt.value} className={`checkout__pay ${data.pagamento === opt.value ? "is-selected" : ""}`}>
                <input
                  type="radio"
                  name="pagamento"
                  value={opt.value}
                  checked={data.pagamento === opt.value}
                  onChange={(e) => update("pagamento", e.target.value)}
                />
                <span>{opt.label}</span>
              </label>
            ))}
          </fieldset>
        )}

        {error && <p className="checkout__error" role="alert">{error}</p>}
      </div>

      <div className="checkout__footer">
        <Button variant="primary" size="lg" fullWidth onClick={next}>
          {step === 2 ? "Revisar pedido" : "Continuar"}
        </Button>
      </div>
    </div>
  );
}

import Button from "../../../components/ui/Button";
import { useCart } from "../../cart/CartContext";
import { WHATSAPP_NUMBER } from "../../../config/business";
import "./ReviewPage.css";

function formatBR(n) { return n.toFixed(2).replace(".", ","); }

export function buildWhatsAppMessage({ items, subtotal, data }) {
  const linhas = [];
  linhas.push("*Novo pedido — Depósito Gelada*");
  linhas.push("");
  linhas.push("*Itens:*");
  items.forEach((i) => {
    linhas.push(`• ${i.qty}x ${i.nome} — R$ ${formatBR(i.qty * i.precoNum)}`);
  });
  linhas.push("");
  linhas.push(`*Subtotal:* R$ ${formatBR(subtotal)}`);
  linhas.push("_(frete a confirmar)_");
  linhas.push("");
  linhas.push("*Entrega:*");
  linhas.push(`Endereço: ${data.endereco}`);
  if (data.referencia) linhas.push(`Referência: ${data.referencia}`);
  linhas.push(`Pagamento: ${data.pagamento}`);
  linhas.push("");
  linhas.push("_Pedido enviado pelo catálogo digital. Aguardo confirmação._");
  return linhas.join("\n");
}

export default function ReviewPage({ data, onBack, onClear }) {
  const { items, subtotal, clear } = useCart();

  const message = buildWhatsAppMessage({ items, subtotal, data });
  const hasNumber = Boolean(WHATSAPP_NUMBER);
  const waUrl = hasNumber
    ? `https://wa.me/${WHATSAPP_NUMBER}?text=${encodeURIComponent(message)}`
    : null;

  function handleSend() {
    if (!hasNumber) return;
    window.open(waUrl, "_blank", "noopener,noreferrer");
    clear();
    onClear?.();
  }

  return (
    <div className="review">
      <header className="review__header">
        <button className="review__back" onClick={onBack} aria-label="Voltar">←</button>
        <h1>Revisar pedido</h1>
      </header>

      <section className="review__card">
        <h2>Itens</h2>
        <ul className="review__items">
          {items.map((i) => (
            <li key={i.nome}>
              <span>{i.qty}x {i.nome}</span>
              <span>R$ {formatBR(i.qty * i.precoNum)}</span>
            </li>
          ))}
        </ul>
        <div className="review__subtotal">
          <span>Subtotal</span>
          <strong>R$ {formatBR(subtotal)}</strong>
        </div>
        <p className="review__note">O frete será confirmado no atendimento. Subtotal não inclui entrega.</p>
      </section>

      <section className="review__card">
        <h2>Entrega</h2>
        <p><strong>Endereço:</strong> {data.endereco}</p>
        {data.referencia && <p><strong>Referência:</strong> {data.referencia}</p>}
        <p><strong>Pagamento:</strong> {data.pagamento}</p>
      </section>

      {!hasNumber && (
        <div className="review__pending" role="alert">
          <strong>WhatsApp não configurado.</strong> Defina <code>VITE_WHATSAPP_NUMBER</code> em <code>.env</code> ou
          edite <code>src/config/business.js</code> com o número do depósito (DDI+DDD+número, só dígitos).
          Enquanto isso, o envio fica desativado.
        </div>
      )}

      <div className="review__actions">
        <Button variant="primary" size="lg" fullWidth onClick={handleSend} disabled={!hasNumber}>
          Enviar pelo WhatsApp
        </Button>
        <Button variant="ghost" fullWidth onClick={onBack}>Voltar e editar</Button>
      </div>
    </div>
  );
}
